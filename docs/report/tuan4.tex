	\chapter{DỊCH VỤ CLOUD ĐƯỢC LỰA CHỌN}

	\section{Tổng quan}

	ChatFrog là chatbot AI tự host: mô hình \tenLLM{} và \tenImage{} chạy trên máy chủ GPU RTX 5090 của nhóm, còn ứng dụng web triển khai trên các dịch vụ cloud dưới đây. Mỗi dịch vụ đảm nhiệm một vai trò riêng và đều là tài nguyên có sẵn hoặc gói miễn phí (Bảng~\ref{tab:dichvu}).

	\begin{table}[H]
		\centering
		\caption{Các dịch vụ cloud sử dụng}\label{tab:dichvu}
		\small
		\begin{tabularx}{\linewidth}{|p{2.8cm}|p{2.2cm}|L|p{2cm}|}
			\hline
			\textbf{Dịch vụ} & \textbf{Mô hình} & \textbf{Vai trò} & \textbf{Chi phí} \\ \hline
			VPS ICTU & Private cloud & Chạy Nginx, container API, lưu file & Có sẵn \\ \hline
			Namecheap & Domain & Tên miền \code{naot.me} & Có sẵn \\ \hline
			Cloudflare & DNSaaS & DNS, proxy ẩn IP, TLS, chống DDoS & Free \\ \hline
			MongoDB Atlas & DBaaS & Lưu người dùng, hội thoại, tin nhắn & Free (M0) \\ \hline
			Tailscale & NaaS & Mạng riêng VPS -- máy chủ GPU & Free \\ \hline
			GitHub & SaaS & Lưu trữ mã nguồn, nguồn triển khai & Free \\ \hline
			Máy chủ GPU RTX 5090 & Self-host & Chạy \tenLLM{} và \tenImage{} & Có sẵn \\ \hline
		\end{tabularx}
	\end{table}

	\section{Thông tin, giao diện và tính năng từng dịch vụ}

	\subsection{VPS ICTU (private cloud)}

	VPS do nhà trường cấp trên hạ tầng private cloud nội bộ: Ubuntu 22.04.5 LTS, 16 vCPU (Intel Xeon Gold 6252), 67 GiB RAM, ổ đĩa 687 GB, IP public \ipvps. Dịch vụ \emph{không có trang web quản trị}; giao diện làm việc là terminal qua SSH bằng khóa công khai (Hình~\ref{fig:vps-console}), màn hình đăng nhập cho biết tải hệ thống, mức dùng ổ đĩa (68,7\%), bộ nhớ (69\%) và các bản cập nhật bảo mật. Nhóm có toàn quyền quản trị nên cài được Nginx, Podman, Node.js, cấu hình tường lửa UFW và lưu file người dùng.

	\anhchup{vps-console.png}{Đăng nhập SSH vào VPS ICTU}{fig:vps-console}

	\subsection{Namecheap -- tên miền}

	Namecheap là nhà đăng ký tên miền; tên miền \code{naot.me} có sẵn, thời hạn 15/05/2026 -- 15/05/2027, bật bảo vệ thông tin chủ sở hữu. Sau khi đăng nhập, menu gồm \emph{Dashboard}, \emph{Domain List}, \emph{SSL Certificates}, \emph{Profile}\ldots; trang chi tiết tên miền (Hình~\ref{fig:namecheap}) có trạng thái, thời hạn, tự động gia hạn và mục \emph{Nameservers}. Nhóm chuyển mục này sang \emph{Custom DNS} với hai nameserver của Cloudflare để Cloudflare quản lý DNS.

	\anhchup{namecheap-ns.png}{Mục Nameservers của naot.me trên Namecheap trỏ về Cloudflare}{fig:namecheap}

	\subsection{Cloudflare (DNSaaS)}

	Cloudflare cung cấp DNS, proxy và lớp bảo vệ ở biên; nhóm dùng gói Free. Dashboard có các nhóm \emph{DNS}, \emph{SSL/TLS}, \emph{Security}, \emph{Caching}, \emph{Analytics}\ldots; trang \emph{DNS $\to$ Records} (Hình~\ref{fig:cf-dns}) liệt kê bản ghi (27/200) với loại, nội dung, trạng thái proxy và TTL. Nhóm dùng: bản ghi \code{A} cho \code{chat} trỏ tới \ipvps{} ở chế độ \emph{Proxied} (ẩn IP thật của VPS), cấp chứng chỉ TLS cho người dùng cuối, bật \emph{Always Use HTTPS} và lọc tấn công cơ bản.

	\anhchup{cloudflare-dns.png}{Bản ghi DNS của naot.me trên Cloudflare}{fig:cf-dns}

	\subsection{MongoDB Atlas (DBaaS)}

	Atlas là MongoDB được quản lý hoàn toàn; nhóm dùng cluster \code{naot} gói Free (M0), MongoDB 8.0.34 trên Google Cloud, tối đa 512 MB. Trang \emph{Project Overview} (Hình~\ref{fig:atlas-console}) hiển thị trạng thái \emph{ACTIVE}, dung lượng đã dùng (103,58/512 MB), số kết nối (6/500), biểu đồ \emph{Throughput} và các nút \emph{Connect}, \emph{Browse collections}, \emph{Metrics}; menu bên trái gồm \emph{Clusters}, \emph{Data Explorer}, \emph{Backup}, \emph{Database \& Network Access}. Nhóm dùng \emph{Database Access} để tạo người dùng CSDL giới hạn quyền, \emph{Network Access} để cấu hình địa chỉ được phép kết nối và \emph{Connect} để lấy chuỗi \code{mongodb+srv://}; kết nối được mã hóa TLS.

	\anhchup{atlas-console.png}{Trang Project Overview của cluster M0 trên MongoDB Atlas}{fig:atlas-console}

	\subsection{Tailscale (NaaS)}

	Tailscale tạo mạng riêng ảo dựa trên WireGuard, tự xuyên NAT và cấp cho mỗi máy một địa chỉ \code{100.x.y.z}. Giao diện có công tắc kết nối, mục \emph{Devices} liệt kê các máy kèm trạng thái và IP, chi tiết mỗi máy gồm tên MagicDNS, IPv4/IPv6 và hạn khóa (Hình~\ref{fig:ts-console}). Máy chủ GPU là \code{ecom-5090} (\code{100.96.27.28}); VPS gọi mô hình qua địa chỉ này mà máy GPU không cần mở cổng hay có IP public.

	\anhchup{tailscale-console.png}{Danh sách thiết bị trong mạng Tailscale}{fig:ts-console}

	\subsection{GitHub (SaaS)}

	GitHub lưu trữ mã nguồn và quản lý phiên bản bằng Git. Mã nguồn ở kho riêng tư \code{chatfrog}, nhánh \code{main}, 11 commit. Trang kho (Hình~\ref{fig:github}) có các thẻ \emph{Code}, \emph{Issues}, \emph{Pull requests}, \emph{Actions}, \emph{Settings}; thẻ \emph{Code} hiển thị cây thư mục (\code{apps}, \code{nginx}, \code{compose.yaml}, \code{Makefile}\ldots) và commit mới nhất. Trên VPS, nhóm lấy bản mới bằng \code{git pull} rồi build lại.

	\anhchup{github-repo.png}{Kho mã nguồn chatfrog trên GitHub}{fig:github}

	\subsection{Máy chủ GPU RTX 5090 (self-host)}

	Máy của nhóm, không phải dịch vụ cloud: Ubuntu 26.04.1 LTS, CPU Intel Core Ultra 9 285K, RAM 128 GiB, GPU NVIDIA RTX 5090 32 GB VRAM; chạy \tenLLM{} qua Ollama / llama.cpp và \tenImage{} qua ComfyUI, chỉ nhận kết nối từ VPS qua Tailscale.

	\section{Tính hợp lý của dịch vụ được lựa chọn}

	\begin{table}[H]
		\centering
		\caption{Lý do chọn dịch vụ}\label{tab:hopli}
		\small
		\begin{tabularx}{\linewidth}{|p{3cm}|L|}
			\hline
			\textbf{Thành phần} & \textbf{Lý do chọn} \\ \hline
			VPS ICTU & Có sẵn, cấu hình lớn (16 vCPU, 67 GiB RAM), IP tĩnh, toàn quyền root \\ \hline
			Namecheap + Cloudflare & Có tên miền thật; Cloudflare cấp TLS, ẩn IP, chống DDoS miễn phí \\ \hline
			MongoDB Atlas & Dữ liệu hội thoại lồng nhau hợp mô hình tài liệu; được quản lý, có dashboard \\ \hline
			Tailscale & Mạng riêng mã hóa, xuyên NAT, cài nhanh; GPU không lộ ra Internet \\ \hline
			GitHub & Quản lý phiên bản mã và cấu hình, triển khai lặp lại bằng \code{git pull} \\ \hline
			GPU tự host & Không trả phí theo token, dữ liệu chat không gửi cho bên thứ ba \\ \hline
		\end{tabularx}
	\end{table}

	\textbf{Ưu điểm:} chi phí gần như bằng 0; bảo mật nhiều lớp (Cloudflare ẩn IP, API chỉ nghe loopback sau Nginx, Atlas dùng TLS, GPU chỉ truy cập qua Tailscale); mỗi dịch vụ một vai trò nên dễ thay thế hoặc mở rộng.

	\textbf{Hạn chế:}
	\begin{itemize}
		\item VPS ICTU chỉ có một máy, không có console để tạo snapshot hay đổi cấu hình, phụ thuộc nhà trường.
		\item Atlas M0 giới hạn 512 MB, không có sao lưu tự động; Network Access đang mở \code{0.0.0.0/0}, an toàn dựa vào mật khẩu mạnh và TLS.
		\item Cloudflare Free không có SLA; đoạn Cloudflare -- VPS đang chạy HTTP (chế độ Flexible).
		\item Máy GPU phụ thuộc đường truyền Tailscale; khi máy tắt thì chat và tạo ảnh không hoạt động.
	\end{itemize}

	Tổ hợp này đáp ứng yêu cầu của đề tài (truy cập qua tên miền và HTTPS, lưu dữ liệu bền vững, gọi mô hình AI trên GPU riêng an toàn) với chi phí gần như bằng không, phù hợp quy mô mini project.

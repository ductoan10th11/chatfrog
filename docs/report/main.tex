	\chapter{MÔ TẢ BÀI TOÁN}
	
	\section{Bài toán}
	
	Dùng dịch vụ AI thương mại gặp hai vấn đề: \textbf{chi phí} cộng dồn theo số thành viên và \textbf{dữ liệu} (tài liệu, ảnh nội bộ) bị gửi ra ngoài. ChatFrog tự host mô hình \tenLLM{} \cite{qwen3} và \tenImage{} trên GPU của nhóm (LLM qua Ollama / llama.cpp \cite{ollama}, mô hình ảnh qua ComfyUI), nên chi phí gần như bằng 0 và dữ liệu nằm trong hạ tầng do nhóm kiểm soát. Người dùng đăng nhập bằng Google, chat với AI, yêu cầu tạo hoặc sửa ảnh, đính kèm tài liệu để AI đọc; quản trị viên duyệt người dùng và theo dõi hệ thống. Đầu vào và đầu ra của hệ thống được tóm tắt trong Bảng~\ref{tab:io}.
	
	\begin{table}[H]
		\centering
		\caption{Đầu vào và đầu ra của hệ thống}\label{tab:io}
		\begin{tabularx}{\linewidth}{|p{2.2cm}|L|}
			\hline
			\textbf{Đầu vào} & Tài khoản Google; tin nhắn văn bản; file đính kèm (ảnh PNG/JPEG/WebP đến 10 MB, tài liệu PDF/DOCX/TXT/CSV đến 20 MB); chỉ dẫn cá nhân \\ \hline
			\textbf{Đầu ra} & Câu trả lời stream theo từng token (SSE); ảnh PNG do mô hình tạo; lịch sử hội thoại; thống kê và trạng thái hệ thống cho quản trị viên \\ \hline

	\subsection{Luồng nghiệp vụ chính}
	
	Luồng chat (Hình~\ref{fig:luong}) gồm các bước: (1) người dùng đăng nhập Google, backend kiểm tra trạng thái đã được duyệt và tạo phiên; (2) người dùng gửi tin nhắn, có thể kèm file; (3) backend lưu tin nhắn vào Atlas, gọi mô hình qua Tailscale; (4) kết quả được stream về trình duyệt theo SSE, đồng thời lưu lại. Với yêu cầu tạo ảnh, backend gọi mô hình ảnh, lưu file PNG lên volume và ghi siêu dữ liệu vào Atlas. Nếu mô hình không phản hồi, tin nhắn được đánh dấu lỗi và người dùng có thể gửi lại.

	\subsection{Đặc tả API}

	\begin{sloppypar}
	Ba endpoint sinh câu trả lời trả về \code{text/event-stream}: các sự kiện \code{reasoning.delta}, \code{text.delta} chứa từng đoạn văn bản, \code{tool.start}/\code{tool.done} báo tiến trình tạo ảnh, \code{queue} báo vị trí chờ GPU, \code{message.done} kết thúc lượt. Lỗi trả về JSON có \code{code} và \code{message}: 401 chưa đăng nhập, 403 chờ duyệt hoặc bị chặn, 409 đang có lượt sinh, 413/415 file quá lớn hoặc sai định dạng; Nginx trả 429 khi đăng nhập quá nhiều lần. Khi mô hình không phản hồi, tin nhắn được lưu ở trạng thái lỗi với mã \code{model_unavailable} (hoặc \code{image_unavailable} cho tạo ảnh).
	\end{sloppypar}

	\section{Thông tin dịch vụ}
	
	Mỗi thành phần dùng một dịch vụ chuyên dụng, đều là tài nguyên có sẵn hoặc gói miễn phí (Bảng~\ref{tab:dichvu}).
	
	\begin{table}[H]
		\centering
		\caption{Các dịch vụ cloud sử dụng}\label{tab:dichvu}
		\begin{tabularx}{\linewidth}{|p{2.8cm}|p{2.3cm}|L|p{2cm}|}
			\hline
			\textbf{Dịch vụ} & \textbf{Mô hình} & \textbf{Vai trò} & \textbf{Chi phí} \\ \hline
			VPS ICTU & IaaS (private cloud) & Chạy Nginx, container API, lưu file & Có sẵn \\ \hline
			Namecheap & Domain & Tên miền \code{naot.me} & Có sẵn \\ \hline
			Cloudflare & DNSaaS & DNS, proxy ẩn IP, TLS, chống DDoS & Free \\ \hline
			MongoDB Atlas & DBaaS & Lưu người dùng, hội thoại, tin nhắn & Free (M0) \\ \hline
			Tailscale & NaaS & Mạng riêng VPS -- máy chủ GPU & Free \\ \hline
			GitHub & SaaS & Lưu trữ mã nguồn, nguồn triển khai & Free \\ \hline

		\end{tabularx}

	\section{Cấu hình tên miền và DNS}\label{sec:dns}
	
	\begin{enumerate}
		\item Cloudflare: \emph{Add a domain} → \code{naot.me} → gói \emph{Free}; ghi lại hai nameserver được cấp.

	\end{enumerate}
	
	Kiểm tra bằng \code{dig +short chat.naot.me}: kết quả phải là IP của Cloudflare (172.67.176.79 và 104.21.88.101), không phải IP của VPS.

	\section{Cấu hình HTTPS với Cloudflare}\label{sec:tls}
	
	Chứng chỉ TLS do Cloudflare tự cấp và quản lý cho \code{chat.naot.me}, nên không cần cài Certbot trên VPS. Với chế độ \emph{Flexible}, kết nối được mã hóa từ trình duyệt tới Cloudflare; từ Cloudflare tới VPS đi qua HTTP cổng 80, vì vậy Nginx chỉ nghe cổng 80. Kiểm tra bằng cách mở \code{http://chat.naot.me}: trình duyệt phải tự chuyển sang HTTPS nhờ \emph{Always Use HTTPS}.

	Hạn chế: đoạn Cloudflare -- VPS chưa được mã hóa. Hướng nâng cấp là cấp \emph{Cloudflare Origin Certificate} cho Nginx và chuyển sang chế độ \emph{Full (strict)}.

	\section{Hướng dẫn sử dụng}
	
	Mở \url{https://chat.naot.me}, chọn \emph{Đăng nhập với Google} (nếu thấy trang chờ duyệt, liên hệ quản trị viên). Nhập câu hỏi rồi nhấn \emph{Enter} (\emph{Shift+Enter} để xuống dòng). Muốn có ảnh, mô tả ảnh cần tạo; muốn sửa, đính kèm ảnh hoặc nhắc tới ảnh vừa tạo. Biểu tượng đính kèm nhận ảnh PNG/JPEG/WebP đến 10 MB và tài liệu PDF/DOCX/TXT/CSV đến 20 MB. Thanh bên dùng để mở lại, ghim, tìm hoặc xóa hội thoại; mục \emph{Cài đặt} để nhập chỉ dẫn cá nhân; quản trị viên dùng mục \emph{Quản trị}.
	
	\chapter{KẾT LUẬN VÀ HƯỚNG PHÁT TRIỂN}
	
	\section{Đánh giá kết quả}
	
	Nhóm đã hoàn thành các mục tiêu đặt ra: ChatFrog chạy trên VPS ICTU, truy cập qua \code{chat.naot.me} với HTTPS; có chat stream, tạo và sửa ảnh, đọc tài liệu, chỉ dẫn cá nhân, đăng nhập Google và trang quản trị. Mô hình chạy trên GPU của nhóm nên không trả phí theo token và dữ liệu không gửi cho bên thứ ba. Chưa đạt: đoạn Cloudflare -- VPS chưa mã hóa, Atlas chưa giới hạn IP, chưa có CI/CD, sao lưu CSDL còn chạy tay (\code{make db-backup}), chưa có dự phòng cho VPS và máy GPU.
	
	\section{Thuận lợi và khó khăn}
	
	\textbf{Thuận lợi:} VPS cấu hình lớn, toàn quyền quản trị; có sẵn GPU RTX 5090 và tên miền; các dịch vụ còn lại có gói miễn phí đủ dùng.
	
	\textbf{Khó khăn và cách khắc phục:}
	\begin{itemize}
		\item \emph{Podman cũ:} \code{apt} của Ubuntu 22.04 chỉ có bản 3.4, thiếu \code{podman compose}; nhóm cài bản mới theo tài liệu chính thức.
		\item \emph{Stream SSE bị giữ lại qua Nginx:} câu trả lời hiện ra một lần; nhóm tắt \code{proxy_buffering}, \code{gzip} cho \code{/api/} và tăng \code{proxy_read_timeout} lên 600 giây.
		\item \emph{Mất IP thật sau Cloudflare:} giới hạn tần suất áp lên IP của Cloudflare; nhóm khai báo \code{set_real_ip_from} và đọc header \code{CF-Connecting-IP}.
		\item \emph{Giới hạn GPU và ngữ cảnh:} backend chỉ chạy 2 lượt chat và 1 lượt tạo ảnh cùng lúc, báo vị trí chờ qua sự kiện \code{queue}; ảnh thu nhỏ về 1024 px, tài liệu cắt theo \code{DOC_MAX_CHARS} để vừa ngữ cảnh 32K token.
	\end{itemize}
	
	\section{Hướng phát triển}
	
	\begin{itemize}
		\item \textbf{Bảo mật:} Cloudflare Origin Certificate với chế độ \emph{Full (strict)}; giới hạn Network Access của Atlas theo IP của VPS; thêm Content-Security-Policy.
		\item \textbf{Tự động hóa và giám sát:} GitHub Actions chạy kiểm thử và triển khai khi đẩy mã; sao lưu CSDL định kỳ; Prometheus và Grafana để theo dõi, cảnh báo.
		\item \textbf{Mở rộng:} thêm máy GPU và cân bằng tải; tìm kiếm ngữ nghĩa (RAG) trên tài liệu dài; chia sẻ và xuất hội thoại.
	\end{itemize}
